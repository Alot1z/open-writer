   289→const uid = () => crypto.randomUUID()
   290→const now = () => new Date().toISOString()
   291→
   292→export const useDataStore = create<DataState>()(
   293→  persist(
   294→    (set, get) => ({
   295→      projects: [],
   296→      chapters: [],
   297→      scenes: [],
   298→      characters: [],
   299→      locations: [],
   300→      notes: [],
   301→      timeline: [],
   302→      world: [],
   303→      objects: [],
   304→      goals: [],
   305→      comments: [],
   306→      versions: [],
   307→      sessions: [],
   308→      relationships: [],
   309→      agentTasks: [],
   310→      backups: [],
   311→
   312→      // ─── Computed Helpers ──────────────────────────────────
   313→
   314→      getProject: (id) => get().projects.find(p => p.id === id),
   315→      getChaptersByProject: (projectId) => get().chapters.filter(c => c.projectId === projectId).sort((a, b) => a.order - b.order),
   316→      getScenesByChapter: (chapterId) => get().scenes.filter(s => s.chapterId === chapterId).sort((a, b) => a.order - b.order),
   317→      getScene: (id) => get().scenes.find(s => s.id === id),
   318→      getChapter: (id) => get().chapters.find(c => c.id === id),
   319→      getCharactersByProject: (projectId) => get().characters.filter(c => c.projectId === projectId),
   320→      getCharacter: (id) => get().characters.find(c => c.id === id),
   321→      getLocationsByProject: (projectId) => get().locations.filter(l => l.projectId === projectId),
   322→      getLocation: (id) => get().locations.find(l => l.id === id),
   323→      getNotesByProject: (projectId) => get().notes.filter(n => n.projectId === projectId),
   324→      getNote: (id) => get().notes.find(n => n.id === id),
   325→      getTimelineByProject: (projectId) => get().timeline.filter(t => t.projectId === projectId),
   326→      getTimelineEvent: (id) => get().timeline.find(t => t.id === id),
   327→      getWorldByProject: (projectId) => get().world.filter(w => w.projectId === projectId),
   328→      getWorldElement: (id) => get().world.find(w => w.id === id),
   329→      getObjectsByProject: (projectId) => get().objects.filter(o => o.projectId === projectId),
   330→      getObject: (id) => get().objects.find(o => o.id === id),
   331→      getGoalsByProject: (projectId) => get().goals.filter(g => g.projectId === projectId),
   332→      getCommentsByProject: (projectId) => get().comments.filter(c => c.projectId === projectId),
   333→      getVersionsByProject: (projectId) => get().versions.filter(v => v.projectId === projectId),
   334→      getSessionsByProject: (projectId) => get().sessions.filter(s => s.projectId === projectId),
   335→      getRelationshipsByProject: (projectId) => get().relationships.filter(r => r.projectId === projectId),
   336→      getBackupsByProject: (projectId) => get().backups.filter(b => b.projectId === projectId),
   337→      getProjectWordCount: (projectId) => {
   338→        const chapters = get().chapters.filter(c => c.projectId === projectId)
   339→        const chapterIds = new Set(chapters.map(c => c.id))
   340→        return get().scenes
   341→          .filter(s => chapterIds.has(s.chapterId))
   342→          .reduce((sum, s) => sum + (s.wordCount || 0), 0)
   343→      },
   344→
   345→      // ─── Project CRUD ─────────────────────────────────────
   346→
   347→      addProject: (data) => {
   348→        const project: Project = {
   349→          id: uid(),
   350→          name: data.name || 'Untitled Project',
   351→          genre: data.genre || '',
   352→          description: data.description || '',
   353→          updatedAt: now(),
   354→          createdAt: now(),
   355→          totalWordCount: 0,
   356→          chapters: [],
   357→          characters: [],
   358→          ...data,
   359→        }
   360→        set(s => ({ projects: [...s.projects, project] }))
   361→        return project
   362→      },
   363→      updateProject: (id, data) => set(s => ({
   364→        projects: s.projects.map(p => p.id === id ? { ...p, ...data, updatedAt: now() } : p)
   365→      })),
   366→      deleteProject: (id) => set(s => ({
   367→        projects: s.projects.filter(p => p.id !== id),
   368→        chapters: s.chapters.filter(c => c.projectId !== id),
   369→        scenes: s.scenes.filter(sc => {
   370→          if (sc.projectId === id) return false
   371→          const chapter = s.chapters.find(c => c.id === sc.chapterId)
   372→          return chapter?.projectId !== id
   373→        }),
   374→        characters: s.characters.filter(c => c.projectId !== id),
   375→        locations: s.locations.filter(l => l.projectId !== id),
   376→        notes: s.notes.filter(n => n.projectId !== id),
   377→        timeline: s.timeline.filter(t => t.projectId !== id),
   378→        world: s.world.filter(w => w.projectId !== id),
   379→        objects: s.objects.filter(o => o.projectId !== id),
   380→        goals: s.goals.filter(g => g.projectId !== id),
   381→        comments: s.comments.filter(c => c.projectId !== id),
   382→        versions: s.versions.filter(v => v.projectId !== id),
   383→        sessions: s.sessions.filter(se => se.projectId !== id),
   384→        relationships: s.relationships.filter(r => r.projectId !== id),
   385→      })),
   386→
   387→      // ─── Chapter CRUD ─────────────────────────────────────
   388→
   389→      addChapter: (data) => {
   390→        const chapter: Chapter = {
   391→          id: uid(),
   392→          projectId: data.projectId || '',
   393→          title: data.title || 'Untitled Chapter',
   394→          order: data.order ?? get().chapters.filter(c => c.projectId === data.projectId).length,
   395→          createdAt: now(),
   396→          updatedAt: now(),
   397→          ...data,
   398→        }
   399→        set(s => ({ chapters: [...s.chapters, chapter] }))
   400→        return chapter
   401→      },
   402→      updateChapter: (id, data) => set(s => ({
   403→        chapters: s.chapters.map(c => c.id === id ? { ...c, ...data, updatedAt: now() } : c)
   404→      })),
   405→      deleteChapter: (id) => {
   406→        const chapter = get().chapters.find(c => c.id === id)
   407→        if (chapter) {
   408→          set(s => ({
   409→            chapters: s.chapters.filter(c => c.id !== id),
   410→            scenes: s.scenes.filter(sc => sc.chapterId !== id),
   411→          }))
   412→        }
   413→      },
   414→
   415→      // ─── Scene CRUD ───────────────────────────────────────
   416→
   417→      addScene: (data) => {
   418→        const scene: Scene = {
   419→          id: uid(),
   420→          chapterId: data.chapterId || '',
   421→          projectId: data.projectId || '',
   422→          title: data.title || 'Untitled Scene',
   423→          content: data.content || '',
   424→          order: data.order ?? get().scenes.filter(s => s.chapterId === data.chapterId).length,
   425→          wordCount: data.wordCount || 0,
   426→          createdAt: now(),
   427→          updatedAt: now(),
   428→          ...data,
   429→        }
   430→        set(s => ({ scenes: [...s.scenes, scene] }))
   431→        return scene
   432→      },
   433→      updateScene: (id, data) => set(s => ({
   434→        scenes: s.scenes.map(sc => sc.id === id ? { ...sc, ...data, updatedAt: now() } : sc)
   435→      })),
   436→      deleteScene: (id) => set(s => ({
   437→        scenes: s.scenes.filter(sc => sc.id !== id),
   438→      })),
   439→
   440→      // ─── Character CRUD ───────────────────────────────────
   441→
   442→      addCharacter: (data) => {
   443→        const character: Character = {
   444→          id: uid(),
   445→          projectId: data.projectId || '',
   446→          name: data.name || 'Unnamed Character',
   447→          role: data.role || 'supporting',
   448→          description: data.description || '',
   449→          motivation: data.motivation || '',
   450→          backstory: data.backstory || '',
   451→          createdAt: now(),
   452→          updatedAt: now(),
   453→          ...data,
   454→        }
   455→        set(s => ({ characters: [...s.characters, character] }))
   456→        return character
   457→      },
   458→      updateCharacter: (id, data) => set(s => ({
   459→        characters: s.characters.map(c => c.id === id ? { ...c, ...data, updatedAt: now() } : c)
   460→      })),
   461→      deleteCharacter: (id) => set(s => ({
   462→        characters: s.characters.filter(c => c.id !== id),
   463→      })),
   464→
   465→      // ─── Location CRUD ────────────────────────────────────
   466→
   467→      addLocation: (data) => {
   468→        const location: Location = {
   469→          id: uid(),
   470→          projectId: data.projectId || '',
   471→          name: data.name || 'Unnamed Location',
   472→          type: data.type || 'setting',
   473→          description: data.description || '',
   474→          atmosphere: data.atmosphere || '',
   475→          createdAt: now(),
   476→          updatedAt: now(),
   477→          ...data,
   478→        }
   479→        set(s => ({ locations: [...s.locations, location] }))
   480→        return location
   481→      },
   482→      updateLocation: (id, data) => set(s => ({
   483→        locations: s.locations.map(l => l.id === id ? { ...l, ...data, updatedAt: now() } : l)
   484→      })),
   485→      deleteLocation: (id) => set(s => ({
   486→        locations: s.locations.filter(l => l.id !== id),
   487→      })),
   488→
   489→      // ─── Note CRUD ────────────────────────────────────────
   490→
   491→      addNote: (data) => {
   492→        const note: Note = {
   493→          id: uid(),
   494→          projectId: data.projectId || '',
   495→          title: data.title || 'Untitled Note',
   496→          content: data.content || '',
   497→          category: data.category || 'general',
   498→          createdAt: now(),
   499→          updatedAt: now(),
   500→          ...data,
   501→        }
   502→        set(s => ({ notes: [...s.notes, note] }))
   503→        return note
   504→      },
   505→      updateNote: (id, data) => set(s => ({
   506→        notes: s.notes.map(n => n.id === id ? { ...n, ...data, updatedAt: now() } : n)
   507→      })),
   508→      deleteNote: (id) => set(s => ({
   509→        notes: s.notes.filter(n => n.id !== id),
   510→      })),

   513→
   514→      addTimelineEvent: (data) => {
   515→        const event: TimelineEvent = {
   516→          id: uid(),

   518→          title: data.title || 'Untitled Event',
   519→          description: data.description || '',
   520→          date: data.date || now(),
   521→          chapterId: data.chapterId || null,
   522→          createdAt: now(),

   525→        }

   529→      updateTimelineEvent: (id, data) => set(s => ({
   530→        timeline: s.timeline.map(t => t.id === id ? { ...t, ...data, updatedAt: now() } : t)

   548→          ...data,

   551→        return element
   552→      },
   553→      updateWorldElement: (id, data) => set(s => ({
   554→        world: s.world.map(w => w.id === id ? { ...w, ...data, updatedAt: now() } : w)
   555→      })),
   556→      deleteWorldElement: (id) => set(s => ({
   557→        world: s.world.filter(w => w.id !== id),
