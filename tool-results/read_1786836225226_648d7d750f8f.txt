   322→const uid = () => crypto.randomUUID()
   323→const now = () => new Date().toISOString()
   324→
   325→export const useDataStore = create<DataState>()(
   326→  persist(
   327→    (set, get) => ({
   328→      projects: [],
   329→      chapters: [],
   330→      scenes: [],
   331→      characters: [],
   332→      locations: [],
   333→      notes: [],
   334→      timeline: [],
   335→      world: [],
   336→      objects: [],
   337→      goals: [],
   338→      comments: [],
   339→      versions: [],
   340→      sessions: [],
   341→      relationships: [],
   342→      agentTasks: [],
   343→      backups: [],
   344→
   345→      // ─── Computed Helpers ──────────────────────────────────
   346→
   347→      getProject: (id) => get().projects.find(p => p.id === id),
   348→      getChaptersByProject: (projectId) => get().chapters.filter(c => c.projectId === projectId).sort((a, b) => a.order - b.order),
   349→      getScenesByChapter: (chapterId) => get().scenes.filter(s => s.chapterId === chapterId).sort((a, b) => a.order - b.order),
   350→      getScene: (id) => get().scenes.find(s => s.id === id),
   351→      getChapter: (id) => get().chapters.find(c => c.id === id),
   352→      getCharactersByProject: (projectId) => get().characters.filter(c => c.projectId === projectId),
   353→      getCharacter: (id) => get().characters.find(c => c.id === id),
   354→      getLocationsByProject: (projectId) => get().locations.filter(l => l.projectId === projectId),
   355→      getLocation: (id) => get().locations.find(l => l.id === id),
   356→      getNotesByProject: (projectId) => get().notes.filter(n => n.projectId === projectId),
   357→      getNote: (id) => get().notes.find(n => n.id === id),
   358→      getTimelineByProject: (projectId) => get().timeline.filter(t => t.projectId === projectId),
   359→      getTimelineEvent: (id) => get().timeline.find(t => t.id === id),
   360→      getWorldByProject: (projectId) => get().world.filter(w => w.projectId === projectId),
   361→      getWorldElement: (id) => get().world.find(w => w.id === id),
   362→      getObjectsByProject: (projectId) => get().objects.filter(o => o.projectId === projectId),
   363→      getObject: (id) => get().objects.find(o => o.id === id),
   364→      getGoalsByProject: (projectId) => get().goals.filter(g => g.projectId === projectId),
   365→      getCommentsByProject: (projectId) => get().comments.filter(c => c.projectId === projectId),
   366→      getVersionsByProject: (projectId) => get().versions.filter(v => v.projectId === projectId),
   367→      getSessionsByProject: (projectId) => get().sessions.filter(s => s.projectId === projectId),
   368→      getRelationshipsByProject: (projectId) => get().relationships.filter(r => r.projectId === projectId),
   369→      getBackupsByProject: (projectId) => get().backups.filter(b => b.projectId === projectId),
   370→      getProjectWordCount: (projectId) => {
   371→        const chapters = get().chapters.filter(c => c.projectId === projectId)
   372→        const chapterIds = new Set(chapters.map(c => c.id))
   373→        return get().scenes
   374→          .filter(s => chapterIds.has(s.chapterId))
   375→          .reduce((sum, s) => sum + (s.wordCount || 0), 0)
   376→      },
   377→
   378→      // ─── Project CRUD ─────────────────────────────────────
   379→
   380→      addProject: (data) => {
   381→        const project: Project = {
   382→          id: uid(),
   383→          name: data.name || 'Untitled Project',
   384→          genre: data.genre || '',
   385→          description: data.description || '',
   386→          updatedAt: now(),
   387→          createdAt: now(),
   388→          totalWordCount: 0,
   389→          chapters: [],
   390→          characters: [],
   391→          ...data,
   392→        }
   393→        set(s => ({ projects: [...s.projects, project] }))
   394→        return project
   395→      },
   396→      updateProject: (id, data) => set(s => ({
   397→        projects: s.projects.map(p => p.id === id ? { ...p, ...data, updatedAt: now() } : p)
   398→      })),
   399→      deleteProject: (id) => set(s => ({
   400→        projects: s.projects.filter(p => p.id !== id),
   401→        chapters: s.chapters.filter(c => c.projectId !== id),
   402→        scenes: s.scenes.filter(sc => {
   403→          if (sc.projectId === id) return false
   404→          const chapter = s.chapters.find(c => c.id === sc.chapterId)
   405→          return chapter?.projectId !== id
   406→        }),
   407→        characters: s.characters.filter(c => c.projectId !== id),
   408→        locations: s.locations.filter(l => l.projectId !== id),
   409→        notes: s.notes.filter(n => n.projectId !== id),
   410→        timeline: s.timeline.filter(t => t.projectId !== id),
   411→        world: s.world.filter(w => w.projectId !== id),
   412→        objects: s.objects.filter(o => o.projectId !== id),
   413→        goals: s.goals.filter(g => g.projectId !== id),
   414→        comments: s.comments.filter(c => c.projectId !== id),
   415→        versions: s.versions.filter(v => v.projectId !== id),
   416→        sessions: s.sessions.filter(se => se.projectId !== id),
   417→        relationships: s.relationships.filter(r => r.projectId !== id),
   418→      })),
   419→
   420→      // ─── Chapter CRUD ─────────────────────────────────────
   421→
   422→      addChapter: (data) => {
   423→        const chapter: Chapter = {
   424→          id: uid(),
   425→          projectId: data.projectId || '',
   426→          title: data.title || 'Untitled Chapter',
   427→          order: data.order ?? get().chapters.filter(c => c.projectId === data.projectId).length,
   428→          createdAt: now(),
   429→          updatedAt: now(),
   430→          ...data,
   431→        }
   432→        set(s => ({ chapters: [...s.chapters, chapter] }))
   433→        return chapter
   434→      },
   435→      updateChapter: (id, data) => set(s => ({
   436→        chapters: s.chapters.map(c => c.id === id ? { ...c, ...data, updatedAt: now() } : c)
   437→      })),
   438→      deleteChapter: (id) => {
   439→        const chapter = get().chapters.find(c => c.id === id)
   440→        if (chapter) {
   441→          set(s => ({
   442→            chapters: s.chapters.filter(c => c.id !== id),
   443→            scenes: s.scenes.filter(sc => sc.chapterId !== id),
   444→          }))
   445→        }
   446→      },
   447→
   448→      // ─── Scene CRUD ───────────────────────────────────────
   449→
   450→      addScene: (data) => {
   451→        const scene: Scene = {
   452→          id: uid(),
   453→          chapterId: data.chapterId || '',
   454→          projectId: data.projectId || '',
   455→          title: data.title || 'Untitled Scene',
   456→          content: data.content || '',
   457→          order: data.order ?? get().scenes.filter(s => s.chapterId === data.chapterId).length,
   458→          wordCount: data.wordCount || 0,
   459→          createdAt: now(),
   460→          updatedAt: now(),
   461→          ...data,
   462→        }
   463→        set(s => ({ scenes: [...s.scenes, scene] }))
   464→        return scene
   465→      },
   466→      updateScene: (id, data) => set(s => ({
   467→        scenes: s.scenes.map(sc => sc.id === id ? { ...sc, ...data, updatedAt: now() } : sc)
   468→      })),
   469→      deleteScene: (id) => set(s => ({
   470→        scenes: s.scenes.filter(sc => sc.id !== id),
   471→      })),
   472→
   473→      // ─── Character CRUD ───────────────────────────────────
   474→
   475→      addCharacter: (data) => {
   476→        const character: Character = {
   477→          id: uid(),
   478→          projectId: data.projectId || '',
   479→          name: data.name || 'Unnamed Character',
   480→          role: data.role || 'supporting',
   481→          description: data.description || '',
   482→          motivation: data.motivation || '',
   483→          backstory: data.backstory || '',
   484→          age: data.age || '',
   485→          occupation: data.occupation || '',
   486→          personality: data.personality || '',
   487→          appearance: data.appearance || '',
   488→          goals: data.goals || '',
   489→          fears: data.fears || '',
   490→          tags: data.tags || '[]',
   491→          createdAt: now(),
   492→          updatedAt: now(),
   493→          ...data,
   494→        }
   495→        set(s => ({ characters: [...s.characters, character] }))
   496→        return character
   497→      },
   498→      updateCharacter: (id, data) => set(s => ({
   499→        characters: s.characters.map(c => c.id === id ? { ...c, ...data, updatedAt: now() } : c)
   500→      })),
   501→      deleteCharacter: (id) => set(s => ({
   502→        characters: s.characters.filter(c => c.id !== id),
   503→      })),
   504→
   505→      // ─── Location CRUD ────────────────────────────────────
   506→
   507→      addLocation: (data) => {
   508→        const location: Location = {
   509→          id: uid(),
   510→          projectId: data.projectId || '',
   511→          name: data.name || 'Unnamed Location',
   512→          type: data.type || 'setting',
   513→          description: data.description || '',
   514→          atmosphere: data.atmosphere || '',
   515→          history: data.history || '',
   516→          features: data.features || '',
   517→          parentLocationId: data.parentLocationId || '',
   518→          tags: data.tags || '[]',
   519→          createdAt: now(),
   520→          updatedAt: now(),
   521→          ...data,
   522→        }
   523→        set(s => ({ locations: [...s.locations, location] }))
   524→        return location
   525→      },
   526→      updateLocation: (id, data) => set(s => ({
   527→        locations: s.locations.map(l => l.id === id ? { ...l, ...data, updatedAt: now() } : l)
   528→      })),
   529→      deleteLocation: (id) => set(s => ({
   530→        locations: s.locations.filter(l => l.id !== id),
   531→      })),
   532→
   533→      // ─── Note CRUD ────────────────────────────────────────
   534→
   535→      addNote: (data) => {
   536→        const note: Note = {
   537→          id: uid(),
   538→          projectId: data.projectId || '',
   539→          title: data.title || 'Untitled Note',
   540→          content: data.content || '',
   541→          category: data.category || 'general',
   542→          linkedType: data.linkedType || '',
   543→          linkedId: data.linkedId || '',
   544→          priority: data.priority ?? 0,
   545→          resolved: data.resolved || false,
   546→          tags: data.tags || '[]',
   547→          createdAt: now(),
   548→          updatedAt: now(),
   549→          ...data,
   550→        }
   551→        set(s => ({ notes: [...s.notes, note] }))
   552→        return note
   553→      },
   554→      updateNote: (id, data) => set(s => ({
   555→        notes: s.notes.map(n => n.id === id ? { ...n, ...data, updatedAt: now() } : n)
   556→      })),
   557→      deleteNote: (id) => set(s => ({
   558→        notes: s.notes.filter(n => n.id !== id),
   559→      })),
   560→
   561→      // ─── Timeline CRUD ────────────────────────────────────
   562→
   563→      addTimelineEvent: (data) => {
   564→        const event: TimelineEvent = {
   565→          id: uid(),
   566→          projectId: data.projectId || '',
   567→          title: data.title || 'Untitled Event',
   568→          description: data.description || '',
   569→          date: data.date || now(),
   570→          chapterId: data.chapterId || null,
   571→          time: data.time || '',
   572→          duration: data.duration || '',
   573→          location: data.location || '',
   574→          characters: data.characters || '[]',
   575→          objects: data.objects || '[]',
   576→          cause: data.cause || '',
   577→          consequence: data.consequence || '',
   578→          eventType: data.eventType || 'custom',
   579→          tags: data.tags || '[]',
   580→          createdAt: now(),
   581→          updatedAt: now(),
   582→          ...data,
   583→        }
   584→        set(s => ({ timeline: [...s.timeline, event] }))
   585→        return event
   586→      },
   587→      updateTimelineEvent: (id, data) => set(s => ({
   588→        timeline: s.timeline.map(t => t.id === id ? { ...t, ...data, updatedAt: now() } : t)
   589→      })),
   590→      deleteTimelineEvent: (id) => set(s => ({
   591→        timeline: s.timeline.filter(t => t.id !== id),
   592→      })),
   593→
   594→      // ─── World CRUD ───────────────────────────────────────
   595→
   596→      addWorldElement: (data) => {
   597→        const element: WorldElement = {
   598→          id: uid(),
   599→          projectId: data.projectId || '',
   600→          name: data.name || 'Unnamed Element',
   601→          category: data.category || 'lore',
   602→          description: data.description || '',
   603→          parent: data.parent || '',
   604→          rules: data.rules || '',
   605→          history: data.history || '',
   606→          tags: data.tags || '[]',
   607→          createdAt: now(),
   608→          updatedAt: now(),
   609→          ...data,
